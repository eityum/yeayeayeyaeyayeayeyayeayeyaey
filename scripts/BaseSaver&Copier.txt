local LPlate, Active, spawnLocation
local function findPlate()
    LPlate, Active, spawnLocation = nil, nil, nil
    for i, v in pairs(WS.Plates:GetChildren()) do
        local o = v:FindFirstChild("Owner")
        if o and o.Value == player then
            LPlate = v:FindFirstChild("Plate")
            Active = v:FindFirstChild("ActiveParts")
            spawnLocation = v:FindFirstChild("SpawnLocation")
            break
        end
    end
end

-- File Management
local FOLDER_NAME = "Blueprints_lua"
local FILE_NAME = FOLDER_NAME .. "/blueprints.json"
local allBlueprints = {}
local selectedSaveName = nil

local function loadSavesFromFile()
    if isfolder and not isfolder(FOLDER_NAME) then
        makefolder(FOLDER_NAME)
    end
    if isfile and readfile and isfile(FILE_NAME) then
        local success, data = pcall(function()
            return HttpService:JSONDecode(readfile(FILE_NAME))
        end)
        if success and type(data) == "table" then
            allBlueprints = data
        end
    end
end

local function saveSavesToFile()
    if writefile then
        task.spawn(function()
            local success, data = pcall(function()
                return HttpService:JSONEncode(allBlueprints)
            end)
            if success then
                writefile(FILE_NAME, data)
            end
        end)
    end
end

loadSavesFromFile()

-- Refresh List Function
local function refreshList()
    for _, child in ipairs(scroll:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end

    local ySize = 0
    for name, data in pairs(allBlueprints) do
        local btn = Instance.new("TextButton", scroll)
        btn.Size = UDim2.new(1, -4, 0, 24)
        btn.BackgroundColor3 = (name == selectedSaveName) and Color3.fromRGB(100, 150, 255) or Color3.fromRGB(40, 40, 40)
        btn.Text = name .. " (" .. #data .. " blocks)"
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        btn.Font = Enum.Font.Gotham
        btn.TextSize = 10
        btn.BorderSizePixel = 0
        Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)
        
        btn.MouseButton1Click:Connect(function()
            selectedSaveName = name
            nameBox.Text = name
            status.Text = "Selected: " .. name
            refreshList()
        end)
        
        ySize = ySize + 28
    end
    scroll.CanvasSize = UDim2.new(0, 0, 0, ySize)
end

-- Save/Copy Logic
local function extractAndSave(targetPlateName, saveName)
    local targetPlateModel = WS.Plates:FindFirstChild(targetPlateName)
    if not targetPlateModel then
        status.Text = "Error: Could not find " .. targetPlateName
        return
    end

    local targetLPlate = targetPlateModel:FindFirstChild("Plate")
    local targetActive = targetPlateModel:FindFirstChild("ActiveParts")
    
    if not targetLPlate or not targetActive then
        status.Text = "Error: Missing plate parts."
        return
    end

    status.Text = "Copying blocks..."
    status.TextColor3 = Color3.fromRGB(255, 200, 50)
    
    local plateCF = targetLPlate.CFrame
    
    task.spawn(function()
        local newBlueprint = {}
        local count = 0

        for _, block in pairs(targetActive:GetChildren()) do
            if block:IsA("Model") and block.PrimaryPart then
                local idObj = block:FindFirstChild("AssetId")
                local rotObj = block:FindFirstChild("NumRotations")
                
                if idObj then
                    local relativeCF = plateCF:ToObjectSpace(block.PrimaryPart.CFrame)
                    table.insert(newBlueprint, {
                        Id = idObj.Value,
                        CFrame = {relativeCF:GetComponents()},
                        Rot = rotObj and rotObj.Value or 0
                    })
                    count = count + 1
                end
            end
        end

        allBlueprints[saveName] = newBlueprint
        selectedSaveName = saveName
        
        saveSavesToFile()
        
        task.defer(function()
            refreshList()
            status.Text = "Copied " .. count .. " blocks from " .. targetPlateName
            status.TextColor3 = Color3.fromRGB(100, 255, 100)
        end)
    end)
end

saveBtn.MouseButton1Click:Connect(function()
    local name = nameBox.Text
    if name == "" then status.Text = "Error: Enter a name." return end

    -- Manual search for your own plate
    local foundPlateName = nil
    for _, plate in ipairs(WS.Plates:GetChildren()) do
        local owner = plate:FindFirstChild("Owner")
        if owner and owner.Value == player then
            foundPlateName = plate.Name
            break
        end
    end

    if foundPlateName then
        extractAndSave(foundPlateName, name)
    else
        status.Text = "Error: Could not find your own plate."
        status.TextColor3 = Color3.fromRGB(255, 100, 100)
    end
end)

copyTargetBtn.MouseButton1Click:Connect(function()
    local targetName = targetBox.Text
    if targetName == "" then status.Text = "Error: Enter target name." return end
    local saveName = nameBox.Text
    if saveName == "" then saveName = targetName .. "'s Base" end

    -- Manual search to bypass the PlotNumber attribute bug
    local foundPlateName = nil
    for _, plate in ipairs(WS.Plates:GetChildren()) do
        local owner = plate:FindFirstChild("Owner")
        if owner and owner.Value and owner.Value.Name == targetName then
            foundPlateName = plate.Name
            break
        end
    end

    if foundPlateName then
        extractAndSave(foundPlateName, saveName)
    else
        status.Text = "Error: Could not find a plate owned by " .. targetName
        status.TextColor3 = Color3.fromRGB(255, 100, 100)
    end
end)

renameBtn.MouseButton1Click:Connect(function()
    local newName = nameBox.Text
    if not selectedSaveName then status.Text = "Error: No save selected." return end
    if newName == "" or newName == selectedSaveName then return end
    allBlueprints[newName] = allBlueprints[selectedSaveName]
    allBlueprints[selectedSaveName] = nil
    selectedSaveName = newName
    saveSavesToFile()
    refreshList()
    status.Text = "Renamed to '" .. newName .. "'."
    status.TextColor3 = Color3.fromRGB(100, 255, 100)
end)

deleteBtn.MouseButton1Click:Connect(function()
    if not selectedSaveName then status.Text = "Error: No save selected." return end
    allBlueprints[selectedSaveName] = nil
    selectedSaveName = nil
    saveSavesToFile()
    refreshList()
    nameBox.Text = ""
    status.Text = "Save deleted."
    status.TextColor3 = Color3.fromRGB(255, 100, 100)
end)

loadBtn.MouseButton1Click:Connect(function()
    if not selectedSaveName or not allBlueprints[selectedSaveName] then
        status.Text = "Error: No save selected."
        return
    end
    findPlate()
    if not LPlate or not Active then status.Text = "Error: Plate not found." return end

    local blueprintData = allBlueprints[selectedSaveName]
    local total = #blueprintData
    
    status.Text = "Loading " .. total .. " blocks..."
    status.TextColor3 = Color3.fromRGB(255, 200, 50)

    local plateCF = LPlate.CFrame
    local currentSpawnLoc = spawnLocation
    
    task.spawn(function()
        local batchSize = 50
        local count = 0
        
        for i = 1, total, batchSize do
            -- Fire a batch of blocks at the exact same time
            for j = i, math.min(i + batchSize - 1, total) do
                local data = blueprintData[j]
                local savedCF = CFrame.new(unpack(data.CFrame))
                local worldCF = plateCF * savedCF
                
                task.spawn(function()
                    pcall(function()
                        StampAsset:InvokeServer(data.Id, worldCF, uuid, {currentSpawnLoc}, data.Rot or 0)
                    end)
                end)
                count = count + 1
            end
            
            -- Wait 0.03 seconds to let the server process the batch
            task.wait(0.03)
            
            -- Update progress
            task.defer(function()
                status.Text = "Loading... " .. count .. " / " .. total
            end)
        end

        task.defer(function()
            status.Text = "Finished loading " .. total .. " blocks."
            status.TextColor3 = Color3.fromRGB(100, 255, 100)
        end)
    end)
end)
